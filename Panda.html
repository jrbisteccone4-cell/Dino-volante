import pygame import random import sys # Inizializzazione Pygame pygame.init() # Dimensioni Schermo WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("MERCEDES HYPERDRIVE: URBAN EVASION") # Colori ROAD_COLOR = (35, 35, 38) GRASS_COLOR = (20, 100, 35) WHITE = (255, 255, 255) YELLOW = (255, 215, 0) RED = (220, 20, 60) ORANGE = (255, 140, 0) DARK_GRAY = (25, 25, 28) CYAN = (0, 230, 255) # Impostazioni Gioco FPS = 60 clock = pygame.time.Clock() # Limiti Pista ROAD_LEFT = 160 ROAD_RIGHT = 640 # Soglia di velocità per esplosione (Valore interno Pygame) SPEED_THRESHOLD = 7.5 def draw_mercedes_car(surface, x, y, width, height, color, is_player=False): """Disegna un'auto dallo stile elegante tipo Mercedes con carrozzeria favolosa.""" ix, iy = int(x), int(y) # Ombra della carrozzeria shadow_surf = pygame.Surface((width + 6, height + 6), pygame.SRCALPHA) pygame.draw.rect(shadow_surf, (0, 0, 0, 80), (0, 0, width + 6, height + 6), border_radius=12) surface.blit(shadow_surf, (ix - 3, iy - 3)) # Corpo principale auto car_rect = pygame.Rect(ix, iy, width, height) pygame.draw.rect(surface, color, car_rect, border_radius=12) # Profilo metallizzato / cromato pygame.draw.rect(surface, (210, 210, 215), car_rect, width=2, border_radius=12) # Parabrezza e Tettuccio glass_color = (120, 210, 255) if is_player else (50, 55, 65) windshield = pygame.Rect(ix + 5, iy + int(height * 0.25), width - 10, int(height * 0.35)) pygame.draw.rect(surface, glass_color, windshield, border_radius=6) pygame.draw.rect(surface, (180, 180, 180), windshield, width=1, border_radius=6) # Griglia Mercedes e Stemma grid_y = iy + height - 8 if is_player else iy + 3 grid_rect = pygame.Rect(ix + 6, grid_y, width - 12, 5) pygame.draw.rect(surface, (220, 220, 220), grid_rect, border_radius=2) # Stemma centrale (Stella Mercedes) center_x = ix + width // 2 center_y = grid_y + 2 pygame.draw.circle(surface, (255, 255, 255), (center_x, center_y), 3) # Fari anteriore e posteriore headlight = (255, 255, 220) taillight = (255, 20, 20) if is_player: pygame.draw.rect(surface, headlight, (ix + 3, iy + 2, 8, 4), border_radius=2) pygame.draw.rect(surface, headlight, (ix + width - 11, iy + 2, 8, 4), border_radius=2) pygame.draw.rect(surface, taillight, (ix + 4, iy + height - 5, 8, 3)) pygame.draw.rect(surface, taillight, (ix + width - 12, iy + height - 5, 8, 3)) else: pygame.draw.rect(surface, headlight, (ix + 3, iy + height - 6, 8, 4), border_radius=2) pygame.draw.rect(surface, headlight, (ix + width - 11, iy + height - 6, 8, 4), border_radius=2) pygame.draw.rect(surface, taillight, (ix + 4, iy + 2, 8, 3)) pygame.draw.rect(surface, taillight, (ix + width - 12, iy + 2, 8, 3)) def draw_explosion(surface, x, y): """Effetto speciale di esplosione in caso di scontro ad alta velocità.""" for _ in range(35): rx = int(x + random.randint(-40, 40)) ry = int(y + random.randint(-40, 40)) radius = random.randint(8, 26) color = random.choice([RED, ORANGE, YELLOW]) pygame.draw.circle(surface, color, (rx, ry), radius) class Player: def __init__(self): self.width = 44 self.height = 80 self.x = float(WIDTH // 2 - self.width // 2) self.y = float(HEIGHT - 120) self.speed = 5.0 self.color = DARK_GRAY def move(self, keys): # Movimento laterale entro i limiti della strada if keys[pygame.K_LEFT] and self.x > ROAD_LEFT + 10: self.x -= 6.0 if keys[pygame.K_RIGHT] and self.x < ROAD_RIGHT - self.width - 10: self.x += 6.0 # Controlli di accelerazione e decelerazione if keys[pygame.K_UP]: self.speed = min(self.speed + 0.1, 13.0) elif keys[pygame.K_DOWN]: self.speed = max(self.speed - 0.15, 2.0) else: if self.speed > 5.0: self.speed -= 0.05 elif self.speed < 5.0: self.speed += 0.05 def draw(self, surface): draw_mercedes_car(surface, self.x, self.y, self.width, self.height, self.color, is_player=True) def get_rect(self): return pygame.Rect(int(self.x), int(self.y), self.width, self.height) class EnemyCar: def __init__(self): self.width = 44 self.height = 80 lanes = [190, 310, 430, 540] self.x = float(random.choice(lanes)) self.y = -100.0 self.base_speed = random.uniform(3.0, 4.5) # Palette colori stile Mercedes self.color = random.choice([ (195, 200, 205), # Argento Iridio (245, 245, 250), # Bianco Polare (20, 35, 65), # Blu Cavansite (75, 10, 20) # Rosso Designo ]) def update(self, player_speed): self.y += self.base_speed + (player_speed - 5.0) def draw(self, surface): draw_mercedes_car(surface, self.x, self.y, self.width, self.height, self.color, is_player=False) def get_rect(self): return pygame.Rect(int(self.x), int(self.y), self.width, self.height) def show_intro_screen(): """Schermata iniziale innovativa con titolo di gioco aggiornato.""" intro = True font_title = pygame.font.SysFont("Arial", 36, bold=True) font_sub = pygame.font.SysFont("Arial", 20, bold=True) while intro: clock.tick(FPS) screen.fill((15, 15, 20)) # Disegno elemento grafico decorativo pygame.draw.rect(screen, CYAN, (WIDTH // 2 - 250, 120, 500, 4), border_radius=2) title_surf = font_title.render("MERCEDES HYPERDRIVE", True, WHITE) subtitle_surf = font_title.render("URBAN EVASION", True, CYAN) start_surf = font_sub.render("PREMI 'SPAZIO' PER AVVIARE I MOTORI", True, YELLOW) info_1 = font_sub.render("• Frecce SU/GIÙ: Accelera / Frena", True, WHITE) info_2 = font_sub.render("• Frecce SINISTRA/DESTRA: Sterza", True, WHITE) info_3 = font_sub.render("• Attento: A bassa velocità eviti l'impatto!", True, (100, 255, 100)) info_4 = font_sub.render("• Ad alta velocità l'impatto causa l'ESPLOSIONE!", True, RED) screen.blit(title_surf, (WIDTH // 2 - title_surf.get_width() // 2, 140)) screen.blit(subtitle_surf, (WIDTH // 2 - subtitle_surf.get_width() // 2, 185)) screen.blit(info_1, (WIDTH // 2 - 200, 270)) screen.blit(info_2, (WIDTH // 2 - 200, 305)) screen.blit(info_3, (WIDTH // 2 - 200, 340)) screen.blit(info_4, (WIDTH // 2 - 200, 375)) screen.blit(start_surf, (WIDTH // 2 - start_surf.get_width() // 2, 470)) pygame.display.flip() for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() if event.type == pygame.KEYDOWN: if event.key == pygame.K_SPACE: intro = False def main(): show_intro_screen() player = Player() enemies = [] spawn_timer = 0 road_offset = 0.0 font = pygame.font.SysFont("Arial", 22, bold=True) game_over = False exploded = False while True: clock.tick(FPS) keys = pygame.key.get_pressed() for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() if game_over and event.type == pygame.KEYDOWN: if event.key == pygame.K_r: main() if not game_over: player.move(keys) # Scorrimento asfalto road_offset = (road_offset + player.speed) % 40 # Generazione auto avversarie spawn_timer += 1 if spawn_timer > 55: enemies.append(EnemyCar()) spawn_timer = 0 player_rect = player.get_rect() # Gestione movimento ed eliminazione bug collisioni for enemy in enemies[:]: enemy.update(player.speed) if enemy.y > HEIGHT + 100: enemies.remove(enemy) continue enemy_rect = enemy.get_rect() if player_rect.colliderect(enemy_rect): if player.speed > SPEED_THRESHOLD: # ALTA VELOCITÀ: Esplosione e Game Over game_over = True exploded = True else: # BASSA VELOCITÀ: Urto controllato (No incidente/esplosione) if player.x < enemy.x: player.x = float(enemy.x - player.width - 4) else: player.x = float(enemy.x + enemy.width + 4) # Mantiene l'auto nei limiti della strada dopo la spinta player.x = max(float(ROAD_LEFT + 10), min(player.x, float(ROAD_RIGHT - player.width - 10))) # Riduzione graduale della velocità post-contatto player.speed = max(2.0, player.speed - 0.4) # --- DISEGNO --- screen.fill(GRASS_COLOR) # Asfalto e Corsie pygame.draw.rect(screen, ROAD_COLOR, (ROAD_LEFT, 0, ROAD_RIGHT - ROAD_LEFT, HEIGHT)) pygame.draw.rect(screen, WHITE, (ROAD_LEFT - 5, 0, 10, HEIGHT)) pygame.draw.rect(screen, WHITE, (ROAD_RIGHT - 5, 0, 10, HEIGHT)) # Strisce tratteggiate for y in range(-40, HEIGHT, 40): pygame.draw.rect(screen, YELLOW, (315, int(y + road_offset), 8, 20)) pygame.draw.rect(screen, YELLOW, (475, int(y + road_offset), 8, 20)) # Auto nemiche for enemy in enemies: enemy.draw(screen) # Disegno Giocatore o Esplosione if not (game_over and exploded): player.draw(screen) else: draw_explosion(screen, player.x + player.width / 2, player.y + player.height / 2) # HUD / Interfaccia utente speed_kmh = int(player.speed * 15) speed_text = font.render(f"Velocità: {speed_kmh} km/h", True, WHITE) screen.blit(speed_text, (20, 20)) if player.speed > SPEED_THRESHOLD: status_txt = font.render("RISCHIO ESPLOSIONE!", True, RED) else: status_txt = font.render("Velocità Sicura", True, (50, 255, 50)) screen.blit(status_txt, (20, 50)) # Schermata di Game Over if game_over: over_font = pygame.font.SysFont("Arial", 32, bold=True) msg = over_font.render("IMPATTO AD ALTA VELOCITÀ!", True, RED) restart_msg = font.render("Premi 'R' per Riavviare", True, WHITE) screen.blit(msg, (WIDTH // 2 - msg.get_width() // 2, HEIGHT // 2 - 30)) screen.blit(restart_msg, (WIDTH // 2 - restart_msg.get_width() // 2, HEIGHT // 2 + 20)) pygame.display.flip() if __name__ == "__main__": main()
